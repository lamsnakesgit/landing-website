### API Контур.Фокус (пример)

```python
import requests

def get_company_taxes(inn: str, api_key: str):
    """Получить налоговые данные компании по ИНН"""
    url = "https://focus-api.kontur.ru/api3/req"
    params = {
        "inn": inn,
        "key": api_key
    }
    response = requests.get(url, params=params)
    data = response.json()
    
    return {
        "inn": inn,
        "name": data.get("name", ""),
        "turnover": data.get("analytics", {}).get("turnover", 0),
        "taxes_year": data.get("analytics", {}).get("taxes", {}).get("year", 0),
        "taxes_quarter": data.get("analytics", {}).get("taxes", {}).get("quarter", 0),
        "employees": data.get("analytics", {}).get("employees", 0),
        "ceo": data.get("ceo", {}).get("name", ""),
        "phone": data.get("contacts", {}).get("phone", "")
    }
```
